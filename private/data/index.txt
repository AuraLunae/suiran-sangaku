<script>
  // URLから動画IDを取り出す(watch / youtu.be / embed / shorts / live に対応)
  function getVideoId(input) {
    let u;
    try { u = new URL(input.trim()); } catch (e) { return null; }

    const host = u.hostname.replace(/^www\.|^m\./, "");
    let id = null;

    if (host === "youtu.be") {
      id = u.pathname.slice(1).split("/")[0];
    } else if (host === "youtube.com" || host === "youtube-nocookie.com") {
      if (u.pathname === "/watch") {
        id = u.searchParams.get("v");
      } else {
        const m = u.pathname.match(/^\/(embed|shorts|live|v)\/([^/?]+)/);
        if (m) id = m[2];
      }
    }
    return id && /^[A-Za-z0-9_-]{11}$/.test(id) ? id : null;
  }

  document.getElementById("form").addEventListener("submit", function (e) {
    e.preventDefault();
    const error = document.getElementById("error");
    const player = document.getElementById("player");
    const id = getVideoId(document.getElementById("url").value);

    error.textContent = "";
    player.innerHTML = "";

    if (!id) {
      player.hidden = true;
      error.textContent = "YouTubeのURLとして認識できません。URLを確認してください。";
      return;
    }

    const iframe = document.createElement("iframe");
    // プライバシー強化モード: youtube-nocookie.com
    iframe.src = "https://www.youtube-nocookie.com/embed/" + id;
    iframe.title = "YouTube動画プレーヤー";
    iframe.allow = "accelerometer; autoplay; clipboard-write; encrypted-media; gyroscope; picture-in-picture; web-share";
    iframe.referrerPolicy = "strict-origin-when-cross-origin";
    iframe.allowFullscreen = true;

    player.appendChild(iframe);
    player.hidden = false;
  });
</script>
